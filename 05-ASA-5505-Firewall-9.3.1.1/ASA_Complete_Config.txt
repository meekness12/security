! =====================================================================
! CISCO PACKET TRACER 9.3.1.1 - CONFIGURING ASA BASIC SETTINGS & FIREWALL
! CISCO ASA 5505 (CCNAS-ASA) - COMPLETE COPY-PASTE CONFIGURATION
! =====================================================================
! Initial Enable Mode: Press [Enter] when prompted for initial password.
! =====================================================================

enable

configure terminal

! ---------------------------------------------------------------------
! Part 2: Basic ASA Settings & Interface Security Levels
! ---------------------------------------------------------------------
hostname CCNAS-ASA
domain-name ccnasecurity.com
enable password ciscoenpa55

! VLAN 1: Inside Interface (LAN 192.168.1.0/24 - Security Level 100)
interface vlan 1
 nameif inside
 ip address 192.168.1.1 255.255.255.0
 security-level 100
 exit

! VLAN 2: Outside Interface (WAN 209.165.200.224/29 - Security Level 0)
interface vlan 2
 nameif outside
 ip address 209.165.200.226 255.255.255.248
 security-level 0
 exit

! ---------------------------------------------------------------------
! Part 3: Routing, Address Translation (PAT), and Inspection Policy
! ---------------------------------------------------------------------
! Static Quad-Zero Default Route via ISP Gateway R1
route outside 0.0.0.0 0.0.0.0 209.165.200.225

! Dynamic PAT: Overload Inside Subnet onto Outside Interface
object network inside-net
 subnet 192.168.1.0 255.255.255.0
 nat (inside,outside) dynamic interface
 exit

! MPF: ICMP Application Inspection to Allow Return Echo Replies
class-map inspection_default
 match default-inspection-traffic
 exit

policy-map global_policy
 class inspection_default
  inspect icmp
 exit

service-policy global_policy global

! ---------------------------------------------------------------------
! Part 4: DHCP Server, AAA Authentication, and Secure Remote Access
! ---------------------------------------------------------------------
! DHCP Daemon for Inside Clients
dhcpd address 192.168.1.5-192.168.1.36 inside
dhcpd dns 209.165.201.2 interface inside
dhcpd enable inside

! Local Authentication Database & AAA for SSH
username admin password adminpa55
aaa authentication ssh console LOCAL

! RSA Key Generation: Enter 'no' if prompted to overwrite existing key
crypto key generate rsa modulus 1024
no

! SSH Access Restrictions & Timeout
ssh 192.168.1.0 255.255.255.0 inside
ssh 172.16.3.3 255.255.255.255 outside
ssh timeout 10

! ---------------------------------------------------------------------
! Part 5: DMZ Interface, Static NAT, and Access Control Lists (ACLs)
! ---------------------------------------------------------------------
! VLAN 3: DMZ Interface (DMZ 192.168.2.0/24 - Security Level 70)
interface vlan 3
 ip address 192.168.2.1 255.255.255.0
 no forward interface vlan 1
 nameif dmz
 security-level 70
 exit

! Assign Physical Port Ethernet 0/2 to DMZ VLAN 3
interface Ethernet0/2
 switchport access vlan 3
 exit

! Static 1:1 NAT: Map Public IP 209.165.200.227 to DMZ Server 192.168.2.3
object network dmz-server
 host 192.168.2.3
 nat (dmz,outside) static 209.165.200.227
 exit

! Inbound ACL Permitting ICMP & HTTP to the Private DMZ Server IP
access-list OUTSIDE-DMZ permit icmp any host 192.168.2.3
access-list OUTSIDE-DMZ permit tcp any host 192.168.2.3 eq 80
access-group OUTSIDE-DMZ in interface outside

! Save Running Configuration to Flash Memory
write memory

! =====================================================================
! End of ASA Configuration
